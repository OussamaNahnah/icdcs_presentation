\begin{frame}{\ModuleFrameTitle{3}{Phase 3: Classification (Criteria)}}

\textbf{How is it calculated?} \\
\textcolor{softgray}{\scriptsize Algorithms are partitioned into equivalence classes by comparing the pre-computed data from \textbf{Synthesis} (syntax) and \textbf{Simulations} (behavior on all grids).}
\vspace{0.4cm}

{\small
\begin{itemize}\setlength\itemsep{0.35cm}
    \item<1-> \textbf{C1: Number of Rules} \textcolor{softgray}{\scriptsize (Syntactic)} \\
          \textcolor{accent}{\scriptsize $\hookrightarrow$ Equal number of rules for each algorithm.}
    \item<2-> \textbf{C2: Rounds in Cycle} \textcolor{softgray}{\scriptsize (Behavioral)} \\
          \textcolor{accent}{\scriptsize $\hookrightarrow$ Same number of execution rounds on all grid simulations.}
    \item<3-> \textbf{C3: Energy in Cycle} \textcolor{softgray}{\scriptsize (Behavioral)} \\
          \textcolor{accent}{\scriptsize $\hookrightarrow$ Same energy cost on all grid simulations (energy = movement or color change).}
    \item<4-> \textbf{C4: Paths in Cycle} \textcolor{softgray}{\scriptsize (Behavioral)} \\
          \textcolor{accent}{\scriptsize $\hookrightarrow$ Identical robot trajectories on all simulations.}
\end{itemize}
}

\end{frame}
